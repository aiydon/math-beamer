% 全局几何: 只在这里改, 页眉 / 正文 / 页脚 / 章节页自动同步
% 必须是长度寄存器 (\newlength): 只有这样 \addtolength{-2\omMargin} 才会做乘法,
% 若写成 \def\omMargin{0.75cm}, "2\omMargin" 会被解析成 "20.75cm"。
% 写内容时注意: columns 各栏宽度之和 + 栏间距 (\columnsep, 默认 10pt) 不能超过
% \textwidth, 否则整块会溢出页边距 (三栏建议每栏 <= 0.31\textwidth)。
\newlength{\omMargin}\setlength{\omMargin}{0.75cm}   % 左右页边距
\newlength{\omLogo}  \setlength{\omLogo}{0.60cm}     % 页眉校徽高度
\newlength{\omBarY}  \setlength{\omBarY}{1.05cm}     % 章节页进度条距页底距离
\newlength{\omBarW}  % 进度条底槽宽度
\def\omCur{}         % 进度条进度对应的帧号
\setbeamersize{text margin left=\omMargin, text margin right=\omMargin}
% 由 \paperwidth 派生的宽度: 在 \begin{document} 时统一算好,
% 避免页眉(先排版)与页脚取到不同的基准
\newlength{\omRuleW}    % 正文宽度 = \textwidth = 左右页边距之间的距离
\newlength{\omHairW}    % 页眉细线中"细"的部分 (另 1.5cm 是品牌蓝短线)
\newlength{\omHeadW}    % 页眉标题可用宽度 (给右侧校徽留出 3.5cm)
\AtBeginDocument{%
    \setlength{\omRuleW}{\textwidth}%
    \setlength{\omHairW}{\omRuleW}\addtolength{\omHairW}{-1.5cm}%
    \setlength{\omHeadW}{\omRuleW}\addtolength{\omHeadW}{-3.5cm}%
    \setlength{\omBarW}{\omRuleW}%
}

\usepackage[SlantFont]{xeCJK}
% 说明: 不启用 xeCJK 的 BoldFont 选项 —— 那是"伪粗体"; 本模板已为中文指定
%       真粗体 (AlibabaPuHuiTi-3-85-Bold), 叠加伪粗体会让笔画糊成黑块。
    % 西文 / 等宽 (FiraCode 是 Fira 家族, 与 Fira Math 同源)
    % FiraCode 只有 Regular/Bold 两个字重, 没有真斜体: 斜体由 AutoFakeSlant 合成。
    % 必须同时声明 ItalicFont / BoldItalicFont, 否则 \textbf{\textit{...}} 会
    % 因 b/it 未定义而悄悄退回直立粗体 (编译时报 Font shape undefined)。
    \defaultfontfeatures[FiraCode]{
        Path           = static/font/,
        Extension      = .ttf,
        UprightFont    = FiraCode-Regular,
        BoldFont       = FiraCode-Bold,
        ItalicFont     = FiraCode-Regular,
        BoldItalicFont = FiraCode-Bold,
        AutoFakeSlant  = 0.2,
        BoldItalicFeatures = {FakeSlant=0.2},   % AutoFakeSlant 只管 m/it, 粗斜体要单独给
        Contextuals    = Alternate
    }
    \setmainfont{FiraCode}
    \setsansfont{FiraCode}
    \setmonofont{FiraCode}
    % 中文 (同样只有 Regular/Bold, 斜体由 AutoFakeSlant 合成)
    \defaultfontfeatures[AlibabaPuHuiTi-3]{
        Path           = static/font/,
        Extension      = .otf,
        UprightFont    = AlibabaPuHuiTi-3-55-Regular,
        BoldFont       = AlibabaPuHuiTi-3-85-Bold,
        ItalicFont     = AlibabaPuHuiTi-3-55-Regular,
        BoldItalicFont = AlibabaPuHuiTi-3-85-Bold,
        AutoFakeSlant  = 0.2,
        BoldItalicFeatures = {FakeSlant=0.2}
    }
    \setCJKmainfont{AlibabaPuHuiTi-3}
    \setCJKsansfont{AlibabaPuHuiTi-3}
    \setCJKmonofont{AlibabaPuHuiTi-3}

\AtBeginSection[]{%
    \begin{frame}[noframenumbering, plain]
        \begin{tikzpicture}[remember picture, overlay]
            % 进度条: 底槽 + 品牌蓝填充 (当前页 / 总页数)
            \edef\omCur{\the\numexpr\value{framenumber}+1\relax}%
            \ifnum\omCur>\inserttotalframenumber\relax
                \edef\omCur{\inserttotalframenumber}%
            \fi
            \pgfmathsetmacro{\omRatio}{\omCur/\inserttotalframenumber}%
            \coordinate (barSW) at (current page.south west);
            \coordinate (barA) at ([xshift=\omMargin, yshift=\omBarY]barSW);
            \coordinate (barNE) at ([xshift=\omBarW, yshift=0.09cm]barA);
            \coordinate (barFG) at ($(barA)!\omRatio!(barNE)$);
            \fill[hair, rounded corners=0.45mm] (barA) rectangle (barNE);
            \fill[sustech, rounded corners=0.45mm] (barA) rectangle (barFG|-barNE);
        \end{tikzpicture}
        \vfill
        \begin{center}
            {\usebeamerfont{part name}\usebeamercolor[fg]{part name}\omSectionLabel\par}%
            \vskip0.20cm
            {\usebeamerfont{section title}\usebeamercolor[fg]{normal text}\insertsection\par}%
            \vskip0.30cm
            {\color{sustech}\rule{1.6cm}{2.4pt}}\par
        \end{center}
        \vfill
    \end{frame}
}

% ------------------------------------------------------------------- 页眉页脚
% 页眉: 标题与正文左对齐; 细线 + 品牌蓝短线; 校徽浮于右上角
% 两个容易踩的坑:
%   1) 页眉排版区**已经在左右页边距之内**, 宽度就是 \textwidth,
%      所以这里不能再用 \hskip\omMargin 平移 (会整体右移一个边距, 且顶到页边);
%   2) 相邻 box 之间必须逐个 \nointerlineskip, 否则 TeX 插入行距,
%      页眉会凭空高出一大截。
\setbeamertemplate{frametitle}{%
    \nointerlineskip
    \vskip0.28cm
    \nointerlineskip
    \hbox to\textwidth{%
        \parbox[b]{\omHeadW}{%
            \usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}\strut\insertframetitle}%
        \hfil
    }%
    \nointerlineskip
    \vskip0.08cm
    \nointerlineskip
    \hbox to\textwidth{%
        \color{sustech}\rule{1.5cm}{1.6pt}%
        \raisebox{0.5pt}{\color{hair}\rule{\omHairW}{0.6pt}}%
    }%
    \vskip0.24cm
}

        \end{tikzpicture}%
    \fi
}
