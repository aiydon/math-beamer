% 章节页: 中文数字标签 + 标题 + 品牌蓝短线 + 页底进度条
\AtBeginSection[]{%
  \begin{frame}[noframenumbering, plain]
    \begin{tikzpicture}[remember picture, overlay]
      \pgfmathsetmacro{\omRatio}{min(\the\numexpr\value{framenumber}+1\relax/\inserttotalframenumber, 1)}
      \coordinate (barA) at ([xshift=\omMargin, yshift=\omBarY]current page.south west);
      \coordinate (barNE) at ([xshift=\textwidth, yshift=0.09cm]barA);
      \coordinate (barFG) at ($(barA)!\omRatio!(barNE)$);
      \fill[hair, rounded corners=0.45mm] (barA) rectangle (barNE);
      \fill[sustech, rounded corners=0.45mm] (barA) rectangle (barFG|-barNE);
    \end{tikzpicture}
    \vfill
    \begin{center}
      {\usebeamerfont{part name}\usebeamercolor[fg]{part name}第\,\zhnum{section}\,节\par}%
      \vskip0.20cm
      {\usebeamerfont{section title}\usebeamercolor[fg]{normal text}\insertsection\par}%
      \vskip0.30cm
      {\color{sustech}\rule{1.6cm}{2.4pt}}\par
    \end{center}
    \vfill
  \end{frame}
}
